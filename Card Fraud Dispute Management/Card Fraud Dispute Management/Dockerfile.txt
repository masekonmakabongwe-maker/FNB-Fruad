# Step 1: Use the heavy SDK image to build and compile the app
FROM mcr.microsoft.com/dotnet/sdk:8.0 AS build
WORKDIR /src

# Copy just the project file first to restore dependencies (caches this layer)
COPY ["Card Fraud Dispute Management.csproj", "./"]
RUN dotnet restore "Card Fraud Dispute Management.csproj"

# Copy the rest of your files and publish the release build
COPY . .
RUN dotnet publish "Card Fraud Dispute Management.csproj" -c Release -o /app/publish /p:UseAppHost=false

# Step 2: Use the lightweight runtime image to actually run the app
FROM mcr.microsoft.com/dotnet/aspnet:8.0 AS final
WORKDIR /app

# Copy the published files from the 'build' stage above
COPY --from=build /app/publish .

# Tell .NET to listen on port 8080 (Standard for .NET 8 container environments)
EXPOSE 8080
ENV ASPNETCORE_HTTP_PORTS=8080

# Start the application
ENTRYPOINT ["dotnet", "Card Fraud Dispute Management.dll"]
